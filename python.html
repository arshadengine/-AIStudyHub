Essential Python for AI & DS Labs

import numpy as np
import pandas as pd

# 1. NumPy Vectorized Broadcasting
x = np.array([1, 2, 3, 4, 5])
y = x * 2 + 10  # Element-wise operations without explicit loops
print("Vectorized result:", y)

# 2. Pandas Aggregation and Imputation
data = {
    'Student_ID': [101, 102, 103, 104],
    'Score': [85, np.nan, 92, 78],
    'Department': ['AI&DS', 'AI&DS', 'CS', 'AI&DS']
}
df = pd.DataFrame(data)

# Impute missing values with mean
df['Score'] = df['Score'].fillna(df['Score'].mean())

# GroupBy calculation
avg_scores = df.groupby('Department')['Score'].mean()
print("\nDepartment Averages:\n", avg_scores)
